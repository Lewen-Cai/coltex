\section{Preliminaries}
\begin{definition}
A function $f\colon \mathbb{R} \to \mathbb{R}$ is \emph{Lipschitz} if there is
a constant $L \ge 0$ with $\lvert f(x) - f(y) \rvert \le L \lvert x - y \rvert$
for all $x, y \in \mathbb{R}$.
\end{definition}

\begin{example}
Every differentiable function with a bounded derivative is Lipschitz.
\end{example}
