\section{Main result}
\begin{definition}
A function $f\colon \mathbb{R} \to \mathbb{R}$ is \emph{Lipschitz} if there is
an $L \ge 0$ with $\lvert f(x) - f(y) \rvert \le L \lvert x - y \rvert$ for all $x, y$.
\end{definition}

\begin{theorem}\label{thm:main}
A Lipschitz function is uniformly continuous.
\end{theorem}

\begin{proof}
Given $\varepsilon > 0$, take $\delta = \varepsilon / L$ (any $\delta$ if $L = 0$).
\end{proof}
